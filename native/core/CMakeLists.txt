cmake_minimum_required(VERSION 3.16)
project(gyro_controller_core VERSION 0.1.0 LANGUAGES CXX)

# 此目标不链接操作系统或 Unreal；Windows 与后续原生传输层可以共用。
add_library(gyro_controller_core STATIC src/controller_protocol.cpp)
target_include_directories(gyro_controller_core PUBLIC include)
target_compile_features(gyro_controller_core PUBLIC cxx_std_17)
set_target_properties(gyro_controller_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(gyro_controller_core PRIVATE /W4 /permissive- /GR- /EHs-c- /utf-8)
else()
    target_compile_options(gyro_controller_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -fno-exceptions -fno-rtti)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(gyro_controller_protocol_tests tests/controller_protocol_tests.cpp)
    target_link_libraries(gyro_controller_protocol_tests PRIVATE gyro_controller_core)
    if(MSVC)
        target_compile_options(gyro_controller_protocol_tests PRIVATE /W4 /utf-8)
    else()
        target_compile_options(gyro_controller_protocol_tests PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti)
    endif()
    add_test(NAME gyro_controller_protocol COMMAND gyro_controller_protocol_tests)
endif()
